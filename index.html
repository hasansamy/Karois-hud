async function loadBriefData() { try { // Cache-busting fetch ensures fresh payload on every reload const res = await fetch('brief.json?v=' + Date.now(), { cache: 'no-store', headers: { 'Cache-Control': 'no-cache' } }); if (!res.ok) throw new Error('brief.json not found'); const data = await res.json(); // 1. Update Wisdom Quote & Author if (data.wisdom || data.quote) { const quoteEl = document.getElementById('brief-quote'); const authorEl = document.getElementById('brief-author'); if (quoteEl) quoteEl.innerText = `"${data.wisdom?.text || data.quote || ''}"`; if (authorEl) authorEl.innerText = `— ${data.wisdom?.author || data.author || 'KAIROS ANCHOR'}`; } // 2. Update High-Impact News / Macro Radar if (data.macro || data.red_folder) { const radarContainer = document.querySelector('.radar-desc'); if (radarContainer) { radarContainer.innerText = data.macro || data.red_folder; } } // 3. Update Weather (if present in brief.json) if (data.weather) { const weatherSub = document.querySelector('.commute-card .commute-info-sub'); const weatherBadge = document.querySelector('.commute-card .commute-badge'); if (weatherSub && data.weather.summary) weatherSub.innerText = data.weather.summary; if (weatherBadge && data.weather.temp) weatherBadge.innerText = data.weather.temp; } // 4. Update Header Subtitle with Synced Date if (data.date) { const subHeader = document.querySelector('.holo-nav .nav-left div div:nth-child(2)'); if (subHeader) { subHeader.innerText = `NODE: SAN JOSE, CA · ${data.date}`; } } } catch (err) { console.warn('Falling back to local cache/defaults:', err); } }
